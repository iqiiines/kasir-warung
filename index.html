import tkinter as tk from tkinter import scrolledtext, messagebox, filedialog import sqlite3 import csv # ========================================== # 1. DATABASE SETUP # ========================================== def setup_database(): conn = sqlite3.connect('warung.db') cursor = conn.cursor() cursor.execute('''CREATE TABLE IF NOT EXISTS produk (id_produk INTEGER PRIMARY KEY AUTOINCREMENT, nama_produk TEXT, harga_jual INTEGER, stok INTEGER)''') conn.commit() conn.close() # ========================================== # 2. LOGIKA APLIKASI # ========================================== def cari_barang(): kata_kunci = entry_cari.get() conn = sqlite3.connect('warung.db') cursor = conn.cursor() cursor.execute("SELECT id_produk, nama_produk, harga_jual, stok FROM produk WHERE nama_produk LIKE ?", ('%' + kata_kunci + '%',)) hasil = cursor.fetchall() conn.close() textbox_hasil.config(state="normal") textbox_hasil.delete("1.0", tk.END) if hasil: for item in hasil: id_p, nama, harga, stok = item total_nilai = harga * stok teks = f"ID: {id_p} | {nama}\n" teks += f"Harga: Rp {harga:,}\n" teks += f"Stok: {stok} pcs\n" teks += f"Total Nilai: Rp {total_nilai:,}\n" teks += f"{'-'*25}\n" textbox_hasil.insert(tk.END, teks) else: textbox_hasil.insert(tk.END, "Barang tidak ditemukan!") textbox_hasil.config(state="disabled") def update_barang(): id_p = entry_id.get() stok_baru = entry_stok_update.get() if not id_p or not stok_baru: messagebox.showerror("Error", "ID dan Stok Baru harus diisi!") return try: conn = sqlite3.connect('warung.db') cursor = conn.cursor() cursor.execute("UPDATE produk SET stok = ? WHERE id_produk = ?", (int(stok_baru), int(id_p))) conn.commit() conn.close() messagebox.showinfo("Sukses", "Stok berhasil diupdate!") cari_barang() # Refresh tampilan entry_id.delete(0, tk.END) entry_stok_update.delete(0, tk.END) except Exception as e: messagebox.showerror("Error", f"Gagal update: {e}") def hapus_barang(): id_p = entry_id.get() if not id_p: messagebox.showerror("Error", "Masukkan ID barang yang ingin dihapus!") return # Konfirmasi popup agar tidak sengaja terhapus yakin = messagebox.askyesno("Konfirmasi Hapus", f"Apakah Anda yakin ingin menghapus barang dengan ID {id_p}?") if yakin: try: conn = sqlite3.connect('warung.db') cursor = conn.cursor() # Cek dulu nama barangnya untuk notifikasi cursor.execute("SELECT nama_produk FROM produk WHERE id_produk = ?", (int(id_p),)) barang = cursor.fetchone() if barang: cursor.execute("DELETE FROM produk WHERE id_produk = ?", (int(id_p),)) conn.commit() messagebox.showinfo("Sukses", f"Barang '{barang[0]}' berhasil dihapus!") cari_barang() # Refresh tampilan entry_id.delete(0, tk.END) entry_stok_update.delete(0, tk.END) else: messagebox.showerror("Error", "ID barang tidak ditemukan!") conn.close() except Exception as e: messagebox.showerror("Error", f"Gagal menghapus: {e}") def tambah_barang(): win = tk.Toplevel(jendela) win.title("Tambah Produk") win.geometry("300x320") tk.Label(win, text="Nama Barang:", font=("Arial", 11)).pack(pady=2) e_nama = tk.Entry(win, font=("Arial", 11)) e_nama.pack(pady=2) tk.Label(win, text="Harga Jual (Rp):", font=("Arial", 11)).pack(pady=2) e_harga = tk.Entry(win, font=("Arial", 11)) e_harga.pack(pady=2) tk.Label(win, text="Stok Awal:", font=("Arial", 11)).pack(pady=2) e_stok = tk.Entry(win, font=("Arial", 11)) e_stok.pack(pady=2) def simpan(): if not e_nama.get() or not e_harga.get() or not e_stok.get(): messagebox.showerror("Error", "Semua kolom harus diisi!") return try: conn = sqlite3.connect('warung.db') cursor = conn.cursor() cursor.execute("INSERT INTO produk (nama_produk, harga_jual, stok) VALUES (?, ?, ?)", (e_nama.get(), int(e_harga.get()), int(e_stok.get()))) conn.commit() conn.close() messagebox.showinfo("Info", "Barang berhasil ditambah!") win.destroy() cari_barang() except ValueError: messagebox.showerror("Error", "Harga dan Stok harus berupa angka!") tk.Button(win, text="Simpan", command=simpan, bg="green", fg="white", font=("Arial", 11, "bold"), width=12).pack(pady=15) # ========================================== # FITUR BARU: EKSPOR DATA KE CSV # ========================================== def ekspor_csv(): conn = sqlite3.connect('warung.db') cursor = conn.cursor() cursor.execute("SELECT id_produk, nama_produk, harga_jual, stok FROM produk") hasil = cursor.fetchall() conn.close() if not hasil: messagebox.showwarning("Peringatan", "Tidak ada data produk untuk diekspor!") return # Membuka jendela dialog untuk memilih lokasi simpan file file_path = filedialog.asksaveasfilename( defaultextension=".csv", filetypes=[("CSV files", "*.csv"), ("All files", "*.*")], title="Simpan Laporan Stok Barang", initialfile="Laporan_Stok_Sembako.csv" ) if file_path: try: with open(file_path, mode='w', newline='', encoding='utf-8') as file: writer = csv.writer(file) # Menulis Header Tabel writer.writerow(['ID Produk', 'Nama Produk', 'Harga Jual', 'Stok Tersisa', 'Total Nilai Aset']) # Menulis Isi Data for row in hasil: id_p, nama, harga, stok = row total_nilai = harga * stok writer.writerow([id_p, nama, harga, stok, total_nilai]) messagebox.showinfo("Sukses", f"Data berhasil diekspor ke:\n{file_path}") except Exception as e: messagebox.showerror("Error", f"Gagal mengekspor data: {e}") # ========================================== # 3. FITUR TAMBAHAN: CHECKOUT (PENJUALAN) # ========================================== def checkout_barang(): id_p = entry_checkout_id.get() jumlah_beli = entry_checkout_qty.get() if not id_p or not jumlah_beli: messagebox.showerror("Error", "ID Barang dan Jumlah Beli harus diisi!") return try: id_p = int(id_p) jumlah_beli = int(jumlah_beli) if jumlah_beli <= 0: messagebox.showerror("Error", "Jumlah beli harus lebih dari 0!") return conn = sqlite3.connect('warung.db') cursor = conn.cursor() # Ambil data produk berdasarkan ID cursor.execute("SELECT nama_produk, harga_jual, stok FROM produk WHERE id_produk = ?", (id_p,)) barang = cursor.fetchone() if not barang: messagebox.showerror("Error", "ID barang tidak ditemukan!") conn.close() return nama, harga, stok_sekarang = barang # Validasi kecukupan stok if stok_sekarang < jumlah_beli: messagebox.showerror("Stok Tidak Cukup", f"Stok '{nama}' hanya tersisa {stok_sekarang} pcs!") conn.close() return # Hitung total belanja total_harga = harga * jumlah_beli stok_baru = stok_sekarang - jumlah_beli # Kurangi stok di database cursor.execute("UPDATE produk SET stok = ? WHERE id_produk = ?", (stok_baru, id_p)) conn.commit() conn.close() # Tampilkan struk/notifikasi transaksi detail_nota = ( f"=== TRANSAKSI BERHASIL ===\n\n" f"Nama Barang : {nama}\n" f"Harga Satuan : Rp {harga:,}\n" f"Jumlah Beli : {jumlah_beli} pcs\n" f"---------------------------\n" f"Total Bayar : Rp {total_harga:,}\n\n" f"Sisa Stok : {stok_baru} pcs" ) messagebox.showinfo("Nota Checkout", detail_nota) # Reset input & refresh tampilan entry_checkout_id.delete(0, tk.END) entry_checkout_qty.delete(0, tk.END) cari_barang() except ValueError: messagebox.showerror("Error", "ID dan Jumlah harus berupa angka bulat!") except Exception as e: messagebox.showerror("Error", f"Terjadi kesalahan: {e}") # ========================================== # 4. TAMPILAN (GUI) # ========================================== setup_database() jendela = tk.Tk() jendela.title("Kasir Warung") jendela.geometry("420x820") # Ukuran jendela diperpanjang sedikit untuk ruang tombol baru # Bagian Atas: Cari, Tambah, Ekspor tk.Label(jendela, text="Cari Nama Barang:", font=("Arial", 12)).pack(pady=5) entry_cari = tk.Entry(jendela, font=("Arial", 12), width=22) entry_cari.pack(pady=2) tk.Button(jendela, text="Cari", command=cari_barang, bg="#4CAF50", fg="white", font=("Arial", 11, "bold"), width=15).pack(pady=4) tk.Button(jendela, text="+ Tambah Produk Baru", command=tambah_barang, bg="orange", fg="white", font=("Arial", 11, "bold"), width=20).pack(pady=4) # Tombol Baru untuk Ekspor tk.Button(jendela, text="📥 Ekspor Data ke Excel (CSV)", command=ekspor_csv, bg="#607D8B", fg="white", font=("Arial", 11, "bold"), width=25).pack(pady=4) # Bagian Tengah: Tempat Hasil Pencarian textbox_hasil = scrolledtext.ScrolledText(jendela, font=("Arial", 11), height=12) textbox_hasil.pack(pady=10, fill=tk.BOTH, expand=True) # Bagian Bawah Panel 1: Checkout (Penjualan) frame_checkout = tk.LabelFrame(jendela, text=" Checkout / Transaksi ", font=("Arial", 10, "bold"), padx=5, pady=8, fg="#2E7D32") frame_checkout.pack(pady=5, fill=tk.X, padx=10) tk.Label(frame_checkout, text="ID:", font=("Arial", 11)).pack(side=tk.LEFT, padx=2) entry_checkout_id = tk.Entry(frame_checkout, width=4, font=("Arial", 11)) entry_checkout_id.pack(side=tk.LEFT, padx=4) tk.Label(frame_checkout, text="Jumlah:", font=("Arial", 11)).pack(side=tk.LEFT, padx=2) entry_checkout_qty = tk.Entry(frame_checkout, width=4, font=("Arial", 11)) entry_checkout_qty.pack(side=tk.LEFT, padx=4) tk.Button(frame_checkout, text="Bayar / Checkout", command=checkout_barang, bg="#2E7D32", fg="white", font=("Arial", 10, "bold")).pack(side=tk.LEFT, padx=6) # Bagian Bawah Panel 2: Kelola (Update & Delete) frame_edit = tk.LabelFrame(jendela, text=" Kelola Data (Masukkan ID Barang) ", font=("Arial", 10, "bold"), padx=5, pady=8) frame_edit.pack(pady=5, fill=tk.X, padx=10) tk.Label(frame_edit, text="ID:", font=("Arial", 11)).pack(side=tk.LEFT, padx=2) entry_id = tk.Entry(frame_edit, width=4, font=("Arial", 11)) entry_id.pack(side=tk.LEFT, padx=4) tk.Label(frame_edit, text="Stok Baru:", font=("Arial", 11)).pack(side=tk.LEFT, padx=2) entry_stok_update = tk.Entry(frame_edit, width=4, font=("Arial", 11)) entry_stok_update.pack(side=tk.LEFT, padx=4) # Tombol Biru untuk Update, Tombol Merah untuk Hapus tk.Button(frame_edit, text="Update", command=update_barang, bg="#2196F3", fg="white", font=("Arial", 10, "bold")).pack(side=tk.LEFT, padx=4) tk.Button(frame_edit, text="Hapus", command=hapus_barang, bg="#F44336", fg="white", font=("Arial", 10, "bold")).pack(side=tk.LEFT, padx=4) jendela.mainloop()
